function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); // ila kan s-satr l-awal khwi, n-katbo fih l-3anawin otomatikiyan if (sheet.getLastRow() === 0) { sheet.appendRow(["Date / Time", "Name", "Phone", "City", "Address", "Offer", "Status"]); } try { var data = JSON.parse(e.postData.contents); var currentDate = Utilities.formatDate(new Date(), "Africa/Casablanca", "yyyy-MM-dd HH:mm:ss"); sheet.appendRow([ currentDate, data.name || "", data.phone || "", data.city || "", data.addr || "بدون عنوان", data.offer || "", "جديد 🆕" ]); return ContentService.createTextOutput(JSON.stringify({"status": "success"})) .setMimeType(ContentService.MimeType.JSON); } catch(error) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.toString()})) .setMimeType(ContentService.MimeType.JSON); } }
